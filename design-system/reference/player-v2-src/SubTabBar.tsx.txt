'use client'

import { cn } from '@/lib/utils'

export interface SubTab {
  id: string
  label: string
  /** Optional count badge next to the label */
  count?: number
}

interface SubTabBarProps {
  tabs: SubTab[]
  active: string
  onChange: (id: string) => void
  className?: string
}

/**
 * Underline-style sub-tab nav. Distinct from the primary pill TabNav so
 * the IA hierarchy reads clearly. Used inside Scouting, Performance,
 * Compare tabs.
 */
export function SubTabBar({ tabs, active, onChange, className }: SubTabBarProps) {
  return (
    <div
      role="tablist"
      className={cn(
        'flex items-center gap-6 border-b border-zinc-200 dark:border-zinc-800',
        className
      )}
    >
      {tabs.map(tab => {
        const isActive = tab.id === active
        return (
          <button
            key={tab.id}
            role="tab"
            type="button"
            aria-selected={isActive}
            onClick={() => onChange(tab.id)}
            className={cn(
              'relative flex items-center gap-2 pb-3 text-sm font-medium transition-colors',
              isActive
                ? '-mb-px border-b-2 border-zinc-950 text-zinc-950 dark:border-zinc-100 dark:text-zinc-100'
                : 'text-zinc-500 hover:text-zinc-900 dark:text-zinc-400 dark:hover:text-zinc-100'
            )}
          >
            {tab.label}
            {typeof tab.count === 'number' && (
              <span
                className={cn(
                  'rounded-sm px-1.5 py-0.5 font-mono text-[10px] font-semibold tabular-nums',
                  isActive
                    ? 'bg-zinc-950 text-white dark:bg-zinc-100 dark:text-zinc-950'
                    : 'bg-zinc-100 text-zinc-700 dark:bg-zinc-800 dark:text-zinc-300'
                )}
              >
                {tab.count}
              </span>
            )}
          </button>
        )
      })}
    </div>
  )
}
