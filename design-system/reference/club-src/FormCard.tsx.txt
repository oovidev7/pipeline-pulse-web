'use client'

import Image from 'next/image'
import Link from 'next/link'
import type { HomeAwaySplit, RecentFormMatch } from '@/lib/types/profiles'
import { Eyebrow } from '@/components/profiles/player-v2/shared/Eyebrow'
import { cn } from '@/lib/utils'

interface FormCardProps {
  recentForm: RecentFormMatch[] | null
  homeAway: HomeAwaySplit | null
  slug: string
}

function chipColor(letter: 'W' | 'D' | 'L'): string {
  if (letter === 'W') return 'bg-emerald-500 text-white'
  if (letter === 'D') return 'bg-zinc-400 text-white'
  return 'bg-red-500 text-white'
}

function VenueSplit({
  label,
  split,
}: {
  label: string
  split: { won: number; drawn: number; lost: number }
}) {
  const total = split.won + split.drawn + split.lost
  const wPct = total > 0 ? (split.won / total) * 100 : 0
  const dPct = total > 0 ? (split.drawn / total) * 100 : 0
  const lPct = total > 0 ? (split.lost / total) * 100 : 0
  return (
    <div className="grid grid-cols-[2.25rem_1fr_auto] items-center gap-2 text-xs">
      <span className="font-mono text-[9px] font-semibold tracking-[0.18em] text-zinc-500 uppercase dark:text-zinc-400">
        {label}
      </span>
      <span className="flex h-1.5 overflow-hidden rounded-full bg-zinc-100 dark:bg-zinc-800">
        <span className="bg-emerald-500" style={{ width: `${wPct}%` }} />
        <span className="bg-zinc-400" style={{ width: `${dPct}%` }} />
        <span className="bg-red-500" style={{ width: `${lPct}%` }} />
      </span>
      <span className="font-mono font-medium text-zinc-900 tabular-nums dark:text-zinc-100">
        {split.won}-{split.drawn}-{split.lost}
      </span>
    </div>
  )
}

function MatchChip({ match, isLatest }: { match: RecentFormMatch; isLatest: boolean }) {
  const tooltip = `${match.isHome ? 'vs' : '@'} ${match.opponentName} (${match.teamScore}-${match.opponentScore})`
  return (
    <div className="flex flex-col items-center gap-1.5">
      <span
        data-testid="form-chip"
        title={tooltip}
        className={cn(
          'flex h-8 w-8 items-center justify-center rounded text-xs font-semibold',
          chipColor(match.result),
          isLatest &&
            'ring-2 ring-emerald-500/70 ring-offset-1 ring-offset-white dark:ring-offset-zinc-900'
        )}
      >
        {match.result}
      </span>
      <div className="flex h-5 w-5 items-center justify-center">
        {match.opponentCrest ? (
          <Image
            src={match.opponentCrest}
            alt={match.opponentName}
            width={20}
            height={20}
            className="h-5 w-5 object-contain"
            title={tooltip}
          />
        ) : (
          <span className="text-[9px] font-medium text-zinc-400 dark:text-zinc-500" title={tooltip}>
            {match.opponentName.slice(0, 3).toUpperCase()}
          </span>
        )}
      </div>
      <span className="font-mono text-[9px] text-zinc-400 tabular-nums dark:text-zinc-500">
        {match.isHome ? 'H' : 'A'}
      </span>
    </div>
  )
}

interface FormSummary {
  pts: number
  maxPts: number
  goalsFor: number
  goalsAgainst: number
  diff: number
}

function summarise(matches: RecentFormMatch[]): FormSummary {
  return matches.reduce<FormSummary>(
    (acc, m) => {
      const pts = m.result === 'W' ? 3 : m.result === 'D' ? 1 : 0
      return {
        pts: acc.pts + pts,
        maxPts: acc.maxPts + 3,
        goalsFor: acc.goalsFor + m.teamScore,
        goalsAgainst: acc.goalsAgainst + m.opponentScore,
        diff: acc.diff + (m.teamScore - m.opponentScore),
      }
    },
    { pts: 0, maxPts: 0, goalsFor: 0, goalsAgainst: 0, diff: 0 }
  )
}

function MidStat({
  label,
  value,
  tone = 'neutral',
}: {
  label: string
  value: string
  tone?: 'up' | 'down' | 'neutral'
}) {
  const toneClass =
    tone === 'up'
      ? 'text-emerald-600 dark:text-emerald-400'
      : tone === 'down'
        ? 'text-red-600 dark:text-red-400'
        : 'text-zinc-950 dark:text-zinc-100'
  return (
    <div className="flex flex-col items-center">
      <span className="font-mono text-[9px] font-semibold tracking-[0.12em] text-zinc-500 uppercase dark:text-zinc-400">
        {label}
      </span>
      <span className={cn('font-outfit text-base font-semibold tabular-nums', toneClass)}>
        {value}
      </span>
    </div>
  )
}

export function FormCard({ recentForm, homeAway, slug }: FormCardProps) {
  const matches = recentForm ?? []
  const hasMatches = matches.length > 0
  const summary = hasMatches ? summarise(matches) : null

  return (
    <section className="flex h-full flex-col rounded-lg border border-zinc-200 bg-white p-4 transition-all hover:border-zinc-400 dark:border-zinc-800 dark:bg-zinc-900 dark:hover:border-zinc-600">
      <header className="mb-3 flex items-baseline justify-between">
        <Eyebrow>Form</Eyebrow>
        <span className="font-mono text-[9px] font-semibold tracking-[0.18em] text-zinc-400 uppercase tabular-nums dark:text-zinc-500">
          Last {hasMatches ? matches.length : 5}
        </span>
      </header>

      {hasMatches ? (
        <div className="mb-4 flex justify-center gap-3">
          {matches.map((m, i) => (
            <Link key={m.matchId} href={`/${slug}/match/${m.matchUuid}`}>
              <MatchChip match={m} isLatest={i === 0} />
            </Link>
          ))}
        </div>
      ) : (
        <div className="mb-4 text-xs text-zinc-400 dark:text-zinc-500">No recent matches</div>
      )}

      {summary ? (
        <div className="mb-4 grid grid-cols-3 gap-2 border-y border-zinc-100 py-3 dark:border-zinc-800">
          <MidStat label="Form Pts" value={`${summary.pts}/${summary.maxPts}`} />
          <MidStat label="Goals" value={`${summary.goalsFor}-${summary.goalsAgainst}`} />
          <MidStat
            label="Diff"
            value={summary.diff > 0 ? `+${summary.diff}` : `${summary.diff}`}
            tone={summary.diff > 0 ? 'up' : summary.diff < 0 ? 'down' : 'neutral'}
          />
        </div>
      ) : null}

      {homeAway ? (
        <div className="mt-auto space-y-2">
          <VenueSplit label="Home" split={homeAway.home} />
          <VenueSplit label="Away" split={homeAway.away} />
        </div>
      ) : null}
    </section>
  )
}
