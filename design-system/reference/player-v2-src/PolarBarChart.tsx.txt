'use client'

import { useMemo, useState } from 'react'
import { motion, AnimatePresence, useReducedMotion } from 'framer-motion'
import { cn } from '@/lib/utils'

export interface PolarBarDataPoint {
  metric: string
  fullName: string
  value: number
  rawValue: number
  percentile: number
  metricKey: string
  isInverted?: boolean
  unit?: string
}

interface PolarBarChartProps {
  data: PolarBarDataPoint[]
  size?: number
  className?: string
  colorScheme?: 'teal' | 'purple' | 'blue'
  showValues?: boolean
  animated?: boolean
  showLegend?: boolean
  /** Entrance: rotate the whole wheel into place once on mount. Opt-in
   * (default off) and automatically disabled under prefers-reduced-motion. */
  spinIn?: boolean
}

function formatPolarValue(value: number, unit?: string): string {
  if (unit === 'km/h' || unit === 'm') return value.toFixed(0)
  // `km` values arrive already converted to kilometres (see buildSkillcornerPizzaData);
  // show 2 d.p. so a ~11 km/90 distance reads "11.03", not "11.0".
  if (unit === 'km') return value.toFixed(2)
  return value.toFixed(1)
}

// Percentile color mapping
function getPercentileColor(percentile: number): string {
  if (percentile >= 90) return 'rgb(34, 197, 94)' // green-500
  if (percentile >= 75) return 'rgb(74, 222, 128)' // green-400
  if (percentile >= 50) return 'rgb(234, 179, 8)' // yellow-500
  if (percentile >= 25) return 'rgb(249, 115, 22)' // orange-500
  return 'rgb(239, 68, 68)' // red-500
}

function getPercentileTier(percentile: number): string {
  if (percentile >= 90) return 'Elite'
  if (percentile >= 75) return 'Above Avg'
  if (percentile >= 50) return 'Average'
  if (percentile >= 25) return 'Below Avg'
  return 'Low'
}

export function PolarBarChart({
  data,
  size = 400,
  className,
  colorScheme: _colorScheme = 'teal',
  showValues: _showValues = true,
  animated = true,
  showLegend = true,
  spinIn = false,
}: PolarBarChartProps) {
  const [hoveredSegment, setHoveredSegment] = useState<number | null>(null)
  const [tooltipPos, setTooltipPos] = useState({ x: 0, y: 0 })
  const reduceMotion = useReducedMotion()

  const center = size / 2
  const outerRadius = size * 0.36
  const innerRadius = size * 0.14
  const labelRadius = size * 0.5

  // The whole wheel (rings, tracks, bars, dividers) spins into place once on
  // mount. Labels/center text stay put and read immediately. Off by default;
  // never runs when the viewer prefers reduced motion.
  const doSpin = spinIn && !reduceMotion

  // Calculate segment angles
  const segments = useMemo(() => {
    const count = data.length
    if (count === 0) return []

    const anglePerSegment = (2 * Math.PI) / count
    const gapAngle = 0.04 // Small gap between segments

    return data.map((item, index) => {
      const startAngle = index * anglePerSegment - Math.PI / 2 + gapAngle / 2
      const endAngle = (index + 1) * anglePerSegment - Math.PI / 2 - gapAngle / 2
      const midAngle = (startAngle + endAngle) / 2

      // Bar length based on percentile (0-100 maps to innerRadius to outerRadius)
      const barRadius = innerRadius + (outerRadius - innerRadius) * (item.percentile / 100)

      return {
        ...item,
        startAngle,
        endAngle,
        midAngle,
        barRadius,
        color: getPercentileColor(item.percentile),
      }
    })
  }, [data, innerRadius, outerRadius])

  // Create arc path
  const createArcPath = (
    innerR: number,
    outerR: number,
    startAngle: number,
    endAngle: number
  ): string => {
    const innerStartX = center + innerR * Math.cos(startAngle)
    const innerStartY = center + innerR * Math.sin(startAngle)
    const innerEndX = center + innerR * Math.cos(endAngle)
    const innerEndY = center + innerR * Math.sin(endAngle)
    const outerStartX = center + outerR * Math.cos(startAngle)
    const outerStartY = center + outerR * Math.sin(startAngle)
    const outerEndX = center + outerR * Math.cos(endAngle)
    const outerEndY = center + outerR * Math.sin(endAngle)

    const largeArcFlag = endAngle - startAngle > Math.PI ? 1 : 0

    return `
      M ${innerStartX} ${innerStartY}
      L ${outerStartX} ${outerStartY}
      A ${outerR} ${outerR} 0 ${largeArcFlag} 1 ${outerEndX} ${outerEndY}
      L ${innerEndX} ${innerEndY}
      A ${innerR} ${innerR} 0 ${largeArcFlag} 0 ${innerStartX} ${innerStartY}
      Z
    `
  }

  // Reference circles for percentile thresholds
  const referenceCircles = [25, 50, 75, 100]

  // Handle segment hover
  const handleSegmentHover = (index: number, event: React.MouseEvent) => {
    setHoveredSegment(index)
    const rect = (event.currentTarget as SVGElement).getBoundingClientRect()
    const svgRect = (event.currentTarget.closest('svg') as SVGElement)?.getBoundingClientRect()
    if (svgRect) {
      setTooltipPos({
        x: rect.x + rect.width / 2 - svgRect.x,
        y: rect.y - svgRect.y - 10,
      })
    }
  }

  // Get label position - horizontal text around the chart
  const getLabelPosition = (midAngle: number, radius: number) => {
    const x = center + radius * Math.cos(midAngle)
    const y = center + radius * Math.sin(midAngle)

    // Determine text anchor based on position
    const angleDeg = (midAngle * 180) / Math.PI + 90 // Normalize to 0-360 from top
    let textAnchor: 'start' | 'middle' | 'end' = 'middle'

    if (angleDeg > 45 && angleDeg < 135) {
      textAnchor = 'start' // Right side
    } else if (angleDeg > 225 && angleDeg < 315) {
      textAnchor = 'end' // Left side
    }

    return { x, y, textAnchor }
  }

  if (data.length === 0) {
    return (
      <div
        className={cn('flex items-center justify-center', className)}
        style={{ width: size, height: size }}
      >
        <p className="text-sm text-slate-500 dark:text-slate-400">No data available</p>
      </div>
    )
  }

  const hoveredData = hoveredSegment !== null ? segments[hoveredSegment] : null

  const padding = size * 0.12
  const totalSize = size + padding * 2

  return (
    <div className={cn('relative', className)}>
      <svg
        width={totalSize}
        height={totalSize}
        viewBox={`${-padding} ${-padding} ${totalSize} ${totalSize}`}
        style={{ overflow: 'visible' }}
      >
        <motion.g
          initial={doSpin ? { rotate: -90, scale: 0.9, opacity: 0 } : false}
          animate={doSpin ? { rotate: 0, scale: 1, opacity: 1 } : undefined}
          transition={doSpin ? { duration: 0.7, ease: [0.22, 1, 0.36, 1] } : undefined}
          style={doSpin ? { transformOrigin: `${center}px ${center}px` } : undefined}
        >
          {/* Background circle */}
          <circle
            cx={center}
            cy={center}
            r={outerRadius}
            fill="none"
            stroke="currentColor"
            strokeWidth="1"
            className="text-slate-200 dark:text-slate-700"
          />

          {/* Reference circles */}
          {referenceCircles.map(pct => {
            const r = innerRadius + (outerRadius - innerRadius) * (pct / 100)
            return (
              <circle
                key={pct}
                cx={center}
                cy={center}
                r={r}
                fill="none"
                stroke="currentColor"
                strokeWidth="0.5"
                strokeDasharray="2,4"
                className="text-slate-200 dark:text-slate-700"
              />
            )
          })}

          {/* Inner circle */}
          <circle
            cx={center}
            cy={center}
            r={innerRadius}
            fill="currentColor"
            className="text-slate-50 dark:text-slate-900"
          />
          <circle
            cx={center}
            cy={center}
            r={innerRadius}
            fill="none"
            stroke="currentColor"
            strokeWidth="1"
            className="text-slate-200 dark:text-slate-700"
          />

          {/* Track segments (background) */}
          {segments.map(seg => (
            <path
              key={`track-${seg.metricKey}`}
              d={createArcPath(innerRadius, outerRadius, seg.startAngle, seg.endAngle)}
              fill="currentColor"
              className="text-slate-100 dark:text-slate-800"
            />
          ))}

          {/* Value segments (bars) with hover interaction */}
          {segments.map((seg, index) => {
            const path = createArcPath(innerRadius, seg.barRadius, seg.startAngle, seg.endAngle)
            const trackPath = createArcPath(innerRadius, outerRadius, seg.startAngle, seg.endAngle)
            const isHovered = hoveredSegment === index

            return (
              <g key={`bar-group-${seg.metricKey}`}>
                {/* Invisible hit area for hover */}
                <path
                  d={trackPath}
                  fill="transparent"
                  className="cursor-pointer"
                  onMouseEnter={e => handleSegmentHover(index, e)}
                  onMouseLeave={() => setHoveredSegment(null)}
                />
                {animated ? (
                  <motion.path
                    d={path}
                    fill={seg.color}
                    initial={{ opacity: 0, scale: 0.8 }}
                    animate={{
                      opacity: 1,
                      scale: isHovered ? 1.05 : 1,
                    }}
                    transition={{
                      delay: animated ? index * 0.05 : 0,
                      duration: 0.3,
                      ease: 'easeOut',
                    }}
                    style={{ transformOrigin: `${center}px ${center}px` }}
                    className="pointer-events-none"
                  />
                ) : (
                  <path
                    d={path}
                    fill={seg.color}
                    className="pointer-events-none"
                    style={{
                      transform: isHovered ? 'scale(1.05)' : 'scale(1)',
                      transformOrigin: `${center}px ${center}px`,
                      transition: 'transform 0.2s ease-out',
                    }}
                  />
                )}
              </g>
            )
          })}

          {/* Divider lines */}
          {segments.map(seg => {
            const x1 = center + innerRadius * Math.cos(seg.startAngle)
            const y1 = center + innerRadius * Math.sin(seg.startAngle)
            const x2 = center + outerRadius * Math.cos(seg.startAngle)
            const y2 = center + outerRadius * Math.sin(seg.startAngle)

            return (
              <line
                key={`line-${seg.metricKey}`}
                x1={x1}
                y1={y1}
                x2={x2}
                y2={y2}
                stroke="currentColor"
                strokeWidth="1.5"
                className="text-white dark:text-slate-900"
              />
            )
          })}
        </motion.g>

        {/* Labels - horizontal text */}
        {segments.map((seg, index) => {
          const { x, y, textAnchor } = getLabelPosition(seg.midAngle, labelRadius)
          const isHovered = hoveredSegment === index

          return (
            <g key={`label-${seg.metricKey}`}>
              {animated ? (
                <>
                  <motion.text
                    x={x}
                    y={y - 6}
                    textAnchor={textAnchor}
                    dominantBaseline="middle"
                    className={cn(
                      'text-[9px] font-semibold tracking-wide uppercase',
                      isHovered
                        ? 'fill-slate-900 dark:fill-white'
                        : 'fill-slate-600 dark:fill-slate-400'
                    )}
                    initial={{ opacity: 0 }}
                    animate={{ opacity: 1 }}
                    transition={{ delay: 0.3 + index * 0.03, duration: 0.3 }}
                  >
                    {seg.metric}
                  </motion.text>
                  <motion.text
                    x={x}
                    y={y + 6}
                    textAnchor={textAnchor}
                    dominantBaseline="middle"
                    className="fill-slate-500 text-[8px] tabular-nums dark:fill-slate-400"
                    initial={{ opacity: 0 }}
                    animate={{ opacity: 1 }}
                    transition={{ delay: 0.35 + index * 0.03, duration: 0.3 }}
                  >
                    {formatPolarValue(seg.rawValue, seg.unit)} {seg.unit ?? 'p90'}
                  </motion.text>
                </>
              ) : (
                <>
                  <text
                    x={x}
                    y={y - 6}
                    textAnchor={textAnchor}
                    dominantBaseline="middle"
                    className={cn(
                      'text-[9px] font-semibold tracking-wide uppercase',
                      isHovered
                        ? 'fill-slate-900 dark:fill-white'
                        : 'fill-slate-600 dark:fill-slate-400'
                    )}
                  >
                    {seg.metric}
                  </text>
                  <text
                    x={x}
                    y={y + 6}
                    textAnchor={textAnchor}
                    dominantBaseline="middle"
                    className="fill-slate-500 text-[8px] tabular-nums dark:fill-slate-400"
                  >
                    {formatPolarValue(seg.rawValue, seg.unit)} {seg.unit ?? 'p90'}
                  </text>
                </>
              )}
            </g>
          )
        })}

        {/* Center text */}
        <text
          x={center}
          y={center - 4}
          textAnchor="middle"
          className="fill-slate-600 text-[7px] font-semibold tracking-wider uppercase dark:fill-slate-300"
        >
          Percentile
        </text>
        <text
          x={center}
          y={center + 6}
          textAnchor="middle"
          className="fill-slate-500 text-[7px] dark:fill-slate-400"
        >
          Rank
        </text>
      </svg>

      {/* Tooltip */}
      <AnimatePresence>
        {hoveredData && (
          <motion.div
            initial={{ opacity: 0, y: 5 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: 5 }}
            transition={{ duration: 0.15 }}
            className="pointer-events-none absolute z-10 rounded-lg border border-slate-200 bg-white/95 px-3 py-2 shadow-lg backdrop-blur-sm dark:border-slate-700 dark:bg-slate-900/95"
            style={{
              left: tooltipPos.x,
              top: tooltipPos.y,
              transform: 'translate(-50%, -100%)',
            }}
          >
            <div className="mb-1.5 text-[10px] font-semibold tracking-wider text-slate-500 uppercase dark:text-slate-400">
              {hoveredData.fullName}
            </div>
            <div className="space-y-1">
              <div className="flex items-center justify-between gap-4">
                <span className="text-[10px] text-slate-500 dark:text-slate-400">
                  {hoveredData.unit ? 'Value' : 'Per 90'}
                </span>
                <span className="font-mono text-xs font-medium text-slate-900 tabular-nums dark:text-white">
                  {hoveredData.rawValue.toFixed(2)}
                  {hoveredData.isInverted && (
                    <span className="ml-1 text-[9px] text-slate-400">(inv)</span>
                  )}
                </span>
              </div>
              <div className="flex items-center justify-between gap-4">
                <span className="text-[10px] text-slate-500 dark:text-slate-400">Percentile</span>
                <span className="text-xs font-bold" style={{ color: hoveredData.color }}>
                  {Math.round(hoveredData.percentile)}th
                  <span className="ml-1 text-[9px] font-normal text-slate-400">
                    ({getPercentileTier(hoveredData.percentile)})
                  </span>
                </span>
              </div>
            </div>
          </motion.div>
        )}
      </AnimatePresence>

      {/* Legend for percentile colors */}
      {showLegend && (
        <div className="mt-4 flex flex-wrap items-center justify-center gap-x-5 gap-y-2 text-[10px] font-medium text-slate-600 dark:text-slate-300">
          <div className="flex items-center gap-1.5">
            <span className="h-2 w-2 rounded-full bg-green-500" />
            <span>Elite (90+)</span>
          </div>
          <div className="flex items-center gap-1.5">
            <span className="h-2 w-2 rounded-full bg-green-400" />
            <span>Above Avg (75-90)</span>
          </div>
          <div className="flex items-center gap-1.5">
            <span className="h-2 w-2 rounded-full bg-yellow-500" />
            <span>Average (50-75)</span>
          </div>
          <div className="flex items-center gap-1.5">
            <span className="h-2 w-2 rounded-full bg-orange-500" />
            <span>Below Avg (25-50)</span>
          </div>
          <div className="flex items-center gap-1.5">
            <span className="h-2 w-2 rounded-full bg-red-500" />
            <span>Low (&lt;25)</span>
          </div>
        </div>
      )}
    </div>
  )
}
