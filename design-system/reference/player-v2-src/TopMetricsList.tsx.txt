'use client'

import { Eyebrow } from '../shared/Eyebrow'
import { StatCard } from '../shared/StatCard'
import { PercentileBar } from '../shared/PercentileBar'
import { getPercentileTextColor } from '../shared/tier-colors'

export interface RankedMetric {
  /** Display name */
  label: string
  /** Optional sub-label, e.g. "(PXT Pass)" */
  subLabel?: string
  /** Raw display value (string already formatted), e.g. "0.18", "34.2 km/h" */
  displayValue: string
  /** 0-100 */
  percentile: number
}

interface TopMetricsListProps {
  title: string
  subtitle?: string
  metrics: RankedMetric[]
  emptyLabel?: string
}

/**
 * Vertical list of metrics ranked by percentile (descending).
 * Used inside the Performance tab below the pizza charts.
 */
export function TopMetricsList({
  title,
  subtitle,
  metrics,
  emptyLabel = 'No data available',
}: TopMetricsListProps) {
  const sorted = [...metrics].sort((a, b) => b.percentile - a.percentile)

  return (
    <StatCard className="p-4">
      <div className="mb-3 flex items-center justify-between">
        <Eyebrow size="sm" dot={false}>
          {title}
        </Eyebrow>
        {subtitle && (
          <div className="font-mono text-[10px] text-zinc-400 dark:text-zinc-500">{subtitle}</div>
        )}
      </div>
      {sorted.length === 0 ? (
        <div className="py-6 text-center text-xs text-zinc-400 dark:text-zinc-500">
          {emptyLabel}
        </div>
      ) : (
        <div className="space-y-2">
          {sorted.map((metric, i) => (
            <div key={metric.label}>
              <div className="flex items-baseline justify-between text-sm">
                <span className="text-zinc-700 dark:text-zinc-300">
                  {metric.label}
                  {metric.subLabel && (
                    <span className="ml-1 text-xs text-zinc-400 dark:text-zinc-500">
                      {metric.subLabel}
                    </span>
                  )}
                </span>
                <span className="font-mono font-semibold text-zinc-950 tabular-nums dark:text-zinc-100">
                  {metric.displayValue} <span className="text-zinc-400 dark:text-zinc-500">·</span>{' '}
                  <span className={getPercentileTextColor(metric.percentile)}>
                    P{Math.round(metric.percentile)}
                  </span>
                </span>
              </div>
              <PercentileBar
                percentile={metric.percentile}
                className="mt-1"
                animate
                delayMs={i * 50}
              />
            </div>
          ))}
        </div>
      )}
    </StatCard>
  )
}
