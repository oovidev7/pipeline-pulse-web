import { AlertTriangle } from 'lucide-react'
import { Eyebrow } from '../shared/Eyebrow'
import { MIN_RELIABLE_POSITION_MINUTES } from '@/lib/services/position-sample'

/**
 * Shown in place of the percentile radar + comparison-bar charts when the
 * selected position's sample is below {@link MIN_RELIABLE_POSITION_MINUTES}.
 * Those charts are percentile-vs-peers, which a handful of matches can't support
 * honestly — so we say so plainly rather than render misleading P-values. The
 * raw match history + season trend below still render (they don't depend on a
 * cohort percentile).
 */
export function InsufficientSampleNotice({
  positionLabel,
  minutes,
}: {
  positionLabel: string
  minutes: number
}) {
  return (
    <div className="flex flex-col items-center justify-center rounded-lg border border-zinc-200 px-6 py-14 text-center dark:border-zinc-800">
      <AlertTriangle className="h-6 w-6 text-amber-500" />
      <div className="mt-3">
        <Eyebrow>Insufficient sample</Eyebrow>
      </div>
      <p className="font-outfit mt-2 text-lg font-semibold text-zinc-900 dark:text-zinc-100">
        Not enough minutes at {positionLabel}
      </p>
      <p className="mt-1.5 max-w-md text-sm text-zinc-500 dark:text-zinc-400">
        <span className="font-mono text-red-600 tabular-nums dark:text-red-400">
          {Math.round(minutes)}&apos;
        </span>{' '}
        in this role is below the{' '}
        <span className="font-mono tabular-nums">{MIN_RELIABLE_POSITION_MINUTES}&apos;</span> needed
        for the radar and comparison charts to be reliable. Pick a position with more minutes to see
        them.
      </p>
    </div>
  )
}
