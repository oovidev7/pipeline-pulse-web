'use client'

import { useMemo, useState } from 'react'
import { ChevronDown } from 'lucide-react'
import { Eyebrow } from '../shared/Eyebrow'
import { StatCard } from '../shared/StatCard'

export interface TrendPoint {
  /** ISO date for X axis ordering */
  date: string
  /** Numeric value for Y axis */
  value: number
  /** Optional label, e.g. "vs Real Madrid" */
  contextLabel?: string
}

export interface TrendSeries {
  id: string
  label: string
  /** Optional Y-axis suffix (e.g. "/90", "km/h") */
  unit?: string
  /** Optional peer-median reference line value */
  peerMedian?: number
  /** Decimal places to show in labels + chart values. Defaults to 2. */
  decimals?: number
  points: TrendPoint[]
}

interface SeasonTrendChartProps {
  series: TrendSeries[]
  /** id of the series to show initially. Defaults to the first series. */
  defaultMetricId?: string
}

const MONTH_INITIALS = [
  'JAN',
  'FEB',
  'MAR',
  'APR',
  'MAY',
  'JUN',
  'JUL',
  'AUG',
  'SEP',
  'OCT',
  'NOV',
  'DEC',
]

/**
 * Simple SVG line chart for the Performance tab. Plots a single metric
 * over time, with a picker so the scout can switch between technical
 * (xG+xA / 90, etc) and physical (top speed, sprints) series without
 * losing context.
 */
export function SeasonTrendChart({ series, defaultMetricId }: SeasonTrendChartProps) {
  // Only consider series that actually have points to plot.
  const available = useMemo(() => series.filter(s => s.points.length > 0), [series])
  const initialId =
    defaultMetricId && available.some(s => s.id === defaultMetricId)
      ? defaultMetricId
      : (available[0]?.id ?? null)
  const [selectedId, setSelectedId] = useState<string | null>(initialId)
  const active = available.find(s => s.id === selectedId) ?? available[0]

  if (!active) return null

  const decimals = active.decimals ?? 2
  // Sort by date ascending
  const sorted = [...active.points].sort(
    (a, b) => new Date(a.date).getTime() - new Date(b.date).getTime()
  )

  const values = sorted.map(p => p.value)
  const maxV = Math.max(...values, active.peerMedian ?? -Infinity)
  const minV = Math.min(...values, active.peerMedian ?? Infinity, 0)
  const range = maxV - minV || 1

  const SVG_W = 800
  const SVG_H = 220
  const PAD_L = 50
  const PAD_R = 20
  const PAD_T = 20
  const PAD_B = 30
  const chartW = SVG_W - PAD_L - PAD_R
  const chartH = SVG_H - PAD_T - PAD_B

  const xFor = (i: number) =>
    sorted.length === 1 ? PAD_L + chartW / 2 : PAD_L + (i / (sorted.length - 1)) * chartW
  const yFor = (v: number) => PAD_T + (1 - (v - minV) / range) * chartH

  const linePath = sorted
    .map((p, i) => `${i === 0 ? 'M' : 'L'}${xFor(i)},${yFor(p.value)}`)
    .join(' ')
  const areaPath = `${linePath} L${xFor(sorted.length - 1)},${PAD_T + chartH} L${xFor(0)},${PAD_T + chartH} Z`

  // Y-axis gridlines (5 lines)
  const gridLines = [0, 0.25, 0.5, 0.75, 1].map(frac => ({
    y: PAD_T + frac * chartH,
    label: (maxV - frac * range).toFixed(decimals),
  }))

  // X-axis labels (start, middle, end)
  const xLabels =
    sorted.length <= 4
      ? sorted.map((p, i) => ({
          x: xFor(i),
          label: MONTH_INITIALS[new Date(p.date).getUTCMonth()],
        }))
      : [
          0,
          Math.floor(sorted.length / 4),
          Math.floor(sorted.length / 2),
          Math.floor((3 * sorted.length) / 4),
          sorted.length - 1,
        ].map(i => ({
          x: xFor(i),
          label: MONTH_INITIALS[new Date(sorted[i].date).getUTCMonth()],
        }))

  const high = Math.max(...values)
  const highIdx = values.indexOf(high)
  const low = Math.min(...values)
  const lowIdx = values.indexOf(low)
  const trendDelta = values[values.length - 1] - values[0]
  const unitSuffix = active.unit ?? ''

  return (
    <div>
      <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
        <Eyebrow>Season trend · {active.label}</Eyebrow>
        {available.length > 1 && (
          <MetricPicker options={available} value={active.id} onChange={id => setSelectedId(id)} />
        )}
      </div>
      <StatCard className="p-5">
        <svg viewBox={`0 0 ${SVG_W} ${SVG_H}`} className="w-full" preserveAspectRatio="none">
          <g stroke="#e4e4e7" strokeWidth={0.5}>
            {gridLines.map((g, i) => (
              <line key={i} x1={PAD_L} y1={g.y} x2={SVG_W - PAD_R} y2={g.y} />
            ))}
          </g>
          <g fontFamily="ui-monospace, monospace" fontSize={10} fill="#a1a1aa">
            {gridLines.map((g, i) => (
              <text key={i} x={PAD_L - 10} y={g.y + 3} textAnchor="end">
                {g.label}
              </text>
            ))}
          </g>
          {active.peerMedian != null && (
            <>
              <line
                x1={PAD_L}
                y1={yFor(active.peerMedian)}
                x2={SVG_W - PAD_R}
                y2={yFor(active.peerMedian)}
                stroke="#a1a1aa"
                strokeWidth={1}
                strokeDasharray="4 4"
              />
              <text
                x={SVG_W - PAD_R}
                y={yFor(active.peerMedian) - 3}
                textAnchor="end"
                fontFamily="ui-monospace, monospace"
                fontSize={9}
                fill="#71717a"
              >
                Peer median · {active.peerMedian.toFixed(decimals)}
              </text>
            </>
          )}
          <defs>
            <linearGradient id="trend-fill" x1="0" x2="0" y1="0" y2="1">
              <stop offset="0%" stopColor="#10b981" stopOpacity={0.12} />
              <stop offset="100%" stopColor="#10b981" stopOpacity={0} />
            </linearGradient>
          </defs>
          <path d={areaPath} fill="url(#trend-fill)" />
          <path d={linePath} fill="none" stroke="#10b981" strokeWidth={2.5} />
          {sorted.map((p, i) => (
            <circle key={i} cx={xFor(i)} cy={yFor(p.value)} r={3} fill="#10b981" />
          ))}
          <g fontFamily="ui-monospace, monospace" fontSize={9} fill="#a1a1aa" textAnchor="middle">
            {xLabels.map((x, i) => (
              <text key={i} x={x.x} y={SVG_H - 8}>
                {x.label}
              </text>
            ))}
          </g>
        </svg>

        <div className="mt-3 flex flex-wrap items-center gap-4 border-t border-zinc-100 pt-3 text-[11px] dark:border-zinc-800/50">
          <div>
            <span className="font-mono text-[9px] tracking-[0.18em] text-zinc-400 uppercase dark:text-zinc-500">
              Season high
            </span>{' '}
            <span className="ml-1 font-mono font-semibold tabular-nums">
              {high.toFixed(decimals)}
              {unitSuffix}
            </span>{' '}
            {sorted[highIdx].contextLabel && (
              <span className="text-zinc-500 dark:text-zinc-400">
                {sorted[highIdx].contextLabel}
              </span>
            )}
          </div>
          <div>
            <span className="font-mono text-[9px] tracking-[0.18em] text-zinc-400 uppercase dark:text-zinc-500">
              Low
            </span>{' '}
            <span className="ml-1 font-mono font-semibold tabular-nums">
              {low.toFixed(decimals)}
              {unitSuffix}
            </span>{' '}
            {sorted[lowIdx].contextLabel && (
              <span className="text-zinc-500 dark:text-zinc-400">
                {sorted[lowIdx].contextLabel}
              </span>
            )}
          </div>
          <div>
            <span className="font-mono text-[9px] tracking-[0.18em] text-zinc-400 uppercase dark:text-zinc-500">
              Trend
            </span>{' '}
            <span
              className={`ml-1 font-mono font-semibold ${
                trendDelta >= 0 ? 'text-emerald-600' : 'text-zinc-700 dark:text-zinc-300'
              }`}
            >
              {trendDelta >= 0 ? '↗ +' : '↘ '}
              {trendDelta.toFixed(decimals)}
              {unitSuffix}
            </span>
          </div>
        </div>
      </StatCard>
    </div>
  )
}

function MetricPicker({
  options,
  value,
  onChange,
}: {
  options: TrendSeries[]
  value: string
  onChange: (id: string) => void
}) {
  return (
    <div className="relative inline-block">
      <select
        value={value}
        onChange={e => onChange(e.target.value)}
        className="appearance-none rounded-md border border-zinc-200 bg-white py-1.5 pr-7 pl-3 font-mono text-[11px] tracking-wider text-zinc-700 uppercase hover:border-zinc-300 focus:border-zinc-400 focus:outline-none dark:border-zinc-800 dark:bg-zinc-900 dark:text-zinc-300 dark:hover:border-zinc-700"
      >
        {options.map(opt => (
          <option key={opt.id} value={opt.id} className="font-sans normal-case">
            {opt.label}
          </option>
        ))}
      </select>
      <ChevronDown className="pointer-events-none absolute top-1/2 right-2 h-3.5 w-3.5 -translate-y-1/2 text-zinc-400 dark:text-zinc-500" />
    </div>
  )
}
